1. TECHNICAL OVERVIEW

This document specifies the architecture, component design, state management,
data models, simulation logic, gamification implementation, audio pipeline,
and quality standards for the "Column Addition and Subtraction within 1000"
interactive lesson module within Intellia SG's Grade 2 Math program.

The module is a React 18 application (Vite + JSX), structured identically to
the reference repository https://github.com/dsamyak/numberbound, and styled
to match https://numberbound.vercel.app/. It will be embedded in WordPress at:

  https://intelliasg.com/courses/grade-1-math/lessons/
  column-addition-subtraction-within-1000/

Audio narration is delivered via ElevenLabs (pre-generated + dynamic fallback)
matching the architecture defined in the reference pipeline document
(Number Bonds Audio & Narration Pipeline).

  return {
    ...state,
    xp: state.xp + xpEarned,
    streak: newStreak,
    maxStreak: Math.max(state.maxStreak, newStreak),
    attemptCount: 0,
    hintsUsed: 0,
  };

  return {
    ...state,
    streak: 0,
    attemptCount: state.attemptCount + 1,
  };

5. QUESTION DATA MODEL

5.1 Question Schema

interface Question {
  id: string;                 // e.g. "Q1_007", "Q9_003"
  type: QuestionType;         // Enum (10 types — see PRD §8.1)
  world: number;              // 0–9 (which world it belongs to)
  difficulty: 1 | 2 | 3;     // 1=easy, 2=medium, 3=hard

  // Operands
  topNumber: number;          // Top row of column (minuend or first addend)
  bottomNumber: number;       // Bottom row (subtrahend or second addend)
  operation: '+' | '-';       // Addition or subtraction
  answer: number;             // Correct answer

  // Place value breakdown (computed)
  topH: number; topT: number; topO: number;
  bottomH: number; bottomT: number; bottomO: number;
  answerH: number; answerT: number; answerO: number;

  // Regrouping flags
  regroup_ones: boolean;       // Ones sum ≥ 10 (addition) or ones top < bottom (sub)
  regroup_tens: boolean;       // Tens sum ≥ 10 (addition) or tens top < bottom (sub)
  carry_value?: number;        // 1 if carry from ones; 1 if carry from tens too
  borrow_tens?: boolean;       // Subtraction: borrow from tens to ones
  borrow_hundreds?: boolean;   // Subtraction: borrow from hundreds to tens

  // Rendering
  questionText: string;        // Full question text (spoken + displayed)
  visual: 'column' | 'word_problem' | 'column_word';
  options: number[];           // MCQ: always 4 options
  correctAnswer: number;

  // Hints
  hint1: string;               // "Check the ones column — do you need to regroup?"
  hint2: string;               // "Let me show you step by step..."
  explanation: string;         // Full explanation shown after 2 failures

  // Word problem extras (Q9–Q10 only)
  characterName?: string;
  objectName?: string;
  scenario?: string;           // Full problem statement
}

type QuestionType =
  | 'col_add_no_regroup'       // Q1
  | 'col_add_regroup_ones'     // Q2
  | 'col_add_regroup_tens'     // Q3
  | 'col_add_regroup_both'     // Q4
  | 'col_sub_no_borrow'        // Q5
  | 'col_sub_borrow_tens'      // Q6
  | 'col_sub_borrow_hundreds'  // Q7
  | 'col_sub_borrow_two'       // Q8
  | 'word_add'                 // Q9
  | 'word_sub';                // Q10

7. COLUMN GRID COMPONENT — ColumnGrid.jsx

// ColumnGrid.jsx — reusable SVG/HTML column algorithm display

const ColumnGrid = ({
  topNumber, bottomNumber, operation,
  answer, missingSlot,  // 'answer' | 'col_ones' | 'col_tens' | 'col_hunds'
  carryOnes, carryTens,
  borrowTens, borrowHunds,
  interactive, onInput,
  size = 'md'           // 'sm' | 'md' | 'lg'
}) => {

  const top = decomposeHTO(topNumber);    // { H, T, O }
  const bot = decomposeHTO(bottomNumber);
  const ans = decomposeHTO(answer);

  return (
    <div className={`col-grid col-grid--${size}`}>
      {/* Column headers */}
      <div className="col-header col-h">H</div>
      <div className="col-header col-t">T</div>
      <div className="col-header col-o">O</div>

      {/* Carry row (appears above top number when carry exists) */}
      {carryOnes && <div className="carry-digit carry-t">1</div>}
      {carryTens  && <div className="carry-digit carry-h">1</div>}

      {/* Top number */}
      <div className="col-cell">{top.H}</div>
      <div className="col-cell">{top.T}</div>
      <div className="col-cell">{top.O}</div>

      {/* Operation symbol + bottom number */}
      <div className="col-op">{operation}</div>
      <div className="col-cell">{bot.H}</div>
      <div className="col-cell">{bot.T}</div>
      <div className="col-cell">{bot.O}</div>

      {/* Divider line */}
      <div className="col-divider" />

      {/* Answer row */}
      <div className={`col-cell ${missingSlot === 'col_hunds' ? 'col-blank' : ''}`}>
        {missingSlot === 'col_hunds' && interactive ? <NumberPad max={9} onSubmit={v => onInput('H', v)} /> : ans.H}
      </div>
      <div className={`col-cell ${missingSlot === 'col_tens' ? 'col-blank' : ''}`}>
        {missingSlot === 'col_tens' && interactive ? <NumberPad max={9} onSubmit={v => onInput('T', v)} /> : ans.T}
      </div>
      <div className={`col-cell ${missingSlot === 'answer' ? 'col-blank' : ''}`}>
        {missingSlot === 'answer' && interactive ? <NumberPad max={999} onSubmit={v => onInput('answer', v)} /> : answer}
      </div>
    </div>
  );
};

// Utility
const decomposeHTO = (n) => ({
  H: Math.floor(n / 100),
  T: Math.floor((n % 100) / 10),
  O: n % 10,
});

9.1 CarryIndicator.jsx

const CarryIndicator = ({ fromColumn, active, onComplete }) => {
  // fromColumn: 'ones' → fly from O to T; 'tens' → fly from T to H
  useEffect(() => {
    if (active) {
      const timer = setTimeout(onComplete, 800);
      return () => clearTimeout(timer);
    }

10. AUDIO IMPLEMENTATION (ElevenLabs Pipeline)

10.1 Audio Engine (src/utils/audio.js)

  The audio engine follows the exact architecture from the Number Bonds
  Audio & Narration Pipeline reference document.

  Segment helper functions:
    say(text)       → { text, style: 'statement' }
    ask(text)       → { text, style: 'question' }
    cheer(text)     → { text, style: 'encouragement' }
    emphasize(text) → { text, style: 'emphasis' }
    think(text)     → { text, style: 'thinking' }
    celebrate(text) → { text, style: 'celebration' }
    instruct(text)  → { text, style: 'instruction' }

  Core function:
    async function getAudioUrl(text, style) {
      // 1. Check audioMap for pre-generated file
      if (audioMap[text]) return audioMap[text];
      // 2. Dynamic fallback: call /api/elevenlabs proxy
      return await fetchElevenLabsAudio(text, style);
    }

    async function narrate(segments) {
      const queueSymbol = Symbol();
      currentQueue = queueSymbol;
      for (let i = 0; i < segments.length; i++) {
        if (currentQueue !== queueSymbol) break;
        const { text, style } = segments[i];
        const url = await getAudioUrl(text, style);
        // Preload next segment while playing current
        if (i + 1 < segments.length) getAudioUrl(segments[i+1].text, segments[i+1].style);
        await playAudio(url);
      }
    }

    function stopNarration() {
      currentQueue = null;
      if (currentAudio) { currentAudio.pause(); currentAudio.currentTime = 0; }
    }

  export default async function handler(req, res) {
    if (req.method !== 'POST') return res.status(405).end();
    const { text, style } = req.body;

    const VOICE_SETTINGS = {
      celebration:          { stability: 0.12, similarity_boost: 0.45, style: 0.75, use_speaker_boost: true },
      encouragement:        { stability: 0.16, similarity_boost: 0.50, style: 0.65, use_speaker_boost: true },
      question:             { stability: 0.20, similarity_boost: 0.55, style: 0.55, use_speaker_boost: true },
      emphasis:             { stability: 0.16, similarity_boost: 0.50, style: 0.60, use_speaker_boost: true },
      thinking:             { stability: 0.24, similarity_boost: 0.60, style: 0.35, use_speaker_boost: true },
      statement:            { stability: 0.20, similarity_boost: 0.55, style: 0.50, use_speaker_boost: true },
      instruction:          { stability: 0.20, similarity_boost: 0.55, style: 0.50, use_speaker_boost: true },
    };

    const response = await fetch(
      'https://api.elevenlabs.io/v1/text-to-speech/Xb7hH8MSUJpSbSDYk0k2',
      {
        method: 'POST',
        headers: {
          'xi-api-key': process.env.ELEVENLABS_API_KEY,
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          text,
          model_id: 'eleven_multilingual_v2',

        }),
      }

    const buffer = await response.arrayBuffer();

    res.send(Buffer.from(buffer));

11. RANDOMISATION ENGINE

11.1 Fisher-Yates Shuffle (utils/shuffle.js)

  export function shuffleArray(array) {
    const arr = [...array];
    for (let i = arr.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [arr[i], arr[j]] = [arr[j], arr[i]];
    }
    return arr;
  }

  export function generateSessionQuestions(bank) {
    // Pick 10 questions per type, ensuring difficulty distribution
    const byType = {};
    bank.forEach(q => {
      if (!byType[q.type]) byType[q.type] = [];
      byType[q.type].push(q);
    });
    const selected = Object.values(byType)
      .flatMap(qs => shuffleArray(qs).slice(0, 10));
    return shuffleArray(selected); // Final session shuffle
  }

11.2 Question Generator (utils/questionGenerator.js)

  // Generates question objects on-the-fly (used if static bank exhausted)
  export function generateColumnAdd(difficulty) {
    const maxH = difficulty === 1 ? 4 : difficulty === 2 ? 7 : 9;
    const h1 = rand(1, maxH), t1 = rand(0, 9), o1 = rand(0, 9);
    const h2 = rand(1, 9 - h1), t2 = rand(0, 9), o2 = rand(0, 9);
    const top = h1 * 100 + t1 * 10 + o1;
    const bot = h2 * 100 + t2 * 10 + o2;
    if (top + bot > 1000) return generateColumnAdd(difficulty); // retry
    const regroup_ones = (o1 + o2) >= 10;
    const onesSum = (o1 + o2) % 10;
    const onesCarry = regroup_ones ? 1 : 0;
    const regroup_tens = (t1 + t2 + onesCarry) >= 10;
    return { topNumber: top, bottomNumber: bot, answer: top + bot,
             regroup_ones, regroup_tens, operation: '+' };
  }

11.3 MCQ Distractor Generation (utils/scoring.js)

  export function generateDistractors(correct, operation, regroup_ones, regroup_tens) {
    const distractors = new Set();
    // Error type 1: forgot to carry ones
    if (regroup_ones) distractors.add(correct - 10);
    // Error type 2: forgot to carry tens
    if (regroup_tens) distractors.add(correct - 100);
    // Error type 3: carried twice (over-eager)
    distractors.add(correct + 10);
    // Error type 4: simple ±random
    distractors.add(correct + 11); distractors.add(correct - 11);

    const filtered = [...distractors]
      .filter(d => d > 0 && d <= 1000 && d !== correct);
    const options = shuffleArray([correct, ...filtered.slice(0, 3)]);
    return options.slice(0, 4);
  }
